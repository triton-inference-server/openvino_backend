# Third-party dependencies of the OpenVINO backend, consumed through the CMake
# dependency provider when configuring with -DTRITON_OPENVINO_USE_CONAN=ON.
# The openvino recipe is exported from conan/openvino by
# cmake/conan_bootstrap.cmake. Keep the version in sync with
# TRITON_BUILD_OPENVINO_VERSION.

[requires]
openvino/2026.4.1
rapidjson/cci.20250205

[options]
# The backend ships libopenvino and its plugins next to libtriton_openvino.so.
openvino/*:shared=True
# The public Triton image runs on Intel CPU only.
openvino/*:enable_gpu=False

[generators]
CMakeDeps
CMakeToolchain
